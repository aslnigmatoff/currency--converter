document.addEventListener('DOMContentLoaded', function() { const convertBtn = document.getElementById('convertBtn'); const amountInput = document.getElementById('amount'); const fromSelect = document.getElementById('from'); const toSelect = document.getElementById('to'); const resultDiv = document.getElementById('result'); // Основной API с поддержкой CNY const API_URL = 'https://api.exchangerate-api.com/v4/latest'; // Фолбэк API const FALLBACK_API = 'https://api.frankfurter.app'; convertBtn.addEventListener('click', async function() { const amount = parseFloat(amountInput.value); const from = fromSelect.value; const to = toSelect.value; if (isNaN(amount) || amount <= 0) { showResult('Введите корректную сумму (больше 0)', 'error'); return; } if (from === to) { showResult(`${amount} ${from} = ${amount} ${to}
Курс: 1 ${from} = 1 ${to}`, 'success'); return; } convertBtn.disabled = true; showResult(' Получаем курс...', ''); try { // Сначала пробуем основной API let convertedAmount = await tryConvert(API_URL, amount, from, to); // Если не получилось, пробуем фолбэк API if (!convertedAmount && from !== 'CNY' && to !== 'CNY') { convertedAmount = await tryConvert(FALLBACK_API, amount, from, to); } if (!convertedAmount) { throw new Error('Не удалось получить курс для выбранных валют'); } showResult( `${amount} ${from} = ${convertedAmount} ${to}`, 'success' ); } catch (error) { console.error('Ошибка:', error); showResult( `Ошибка: ${error.message || 'Сервис временно недоступен'}
Попробуйте другие валюты или повторите позже`, 'error' ); } finally { convertBtn.disabled = false; } }); async function tryConvert(apiUrl, amount, from, to) { try { let url; if (apiUrl.includes('exchangerate-api')) { // Для exchangerate-api.com url = `${apiUrl}/${from}`; } else { // Для frankfurter.app url = `${apiUrl}/latest?from=${from}`; } const response = await fetch(url); if (!response.ok) throw new Error('API не ответил'); const data = await response.json(); // Прямой доступ к курсу const rate = data.rates && data.rates[to]; if (!rate) { throw new Error('Нет данных о курсе'); } return (amount * rate).toFixed(2); } catch (error) { console.warn(`Ошибка в ${apiUrl}:`, error); return null; } } function showResult(message, type) { resultDiv.innerHTML = message; resultDiv.className = type; } });
